    46→function deriveOwner(
    47→  args: any[],
    48→  session: SessionData | null,
    49→  idx: number
    50→): string {
    51→  const fromArgs = args[idx];
    52→  if (typeof fromArgs === "string" && fromArgs.length > 0) return fromArgs;
    53→  if (session) {
    54→    if (session.activeUsername) return session.activeUsername;
    55→    if (session.username) return session.username;
    56→  }
    57→  return "";
    58→}

    61→  args: any[],
    62→  session: SessionData | null,
    63→  idx: number
    64→): string {
    65→  const fromArgs = args[idx];
    66→  if (typeof fromArgs === "string" && fromArgs.length > 0) return fromArgs;
    67→  if (session) {
    68→    if (session.activeRole) return session.activeRole;
    69→    if (session.role) return session.role;
    70→  }
    71→  return "user";

   144→// function getInputQC(ownerUsername, role, filter)
   145→// filter: {paramID, lotID, paramIDs[], bidang, namaAlat, startDate, endDate, year}
   146→// Returns array of QC objects. superadmin sees all.
   147→export async function getInputQC(args: any[], session: SessionData | null) {
   148→  const ownerUsername = deriveOwner(args, session, 0);
   149→  const role = deriveRole(args, session, 1);
   150→  const filter = args[2] || {};
   151→  try {
   152→    const where: any = {};
   153→    if (role !== "superadmin") where.ownerUsername = ownerUsername;
   154→    if (filter.paramID) where.paramID = String(filter.paramID);
   155→    if (filter.lotID) where.lotID = String(filter.lotID);
   156→    if (filter.paramIDs && Array.isArray(filter.paramIDs) && filter.paramIDs.length > 0) {
   157→      where.paramID = { in: filter.paramIDs.map(String) };
   158→    }
   159→
   160→    // Date-range filter on tanggal (YYYY-MM-DD string in DB)
   161→    if (filter.startDate || filter.endDate) {
   162→      const dateFilter: any = {};
   163→      if (filter.startDate) {
   164→        const sd = parseDateStr(String(filter.startDate));
   165→        if (sd) {
   166→          const iso = dateToISO(sd);
   167→          if (iso) dateFilter.gte = iso;
   168→        }
   169→      }
   170→      if (filter.endDate) {
   171→        const ed = parseDateStr(String(filter.endDate));
   172→        if (ed) {
   173→          const iso = dateToISO(ed);
   174→          if (iso) dateFilter.lte = iso;
   175→        }
   176→      }
   177→      if (Object.keys(dateFilter).length > 0) where.tanggal = dateFilter;
   178→    }
   179→
   180→    // Year filter — in GAS this selects a different sheet (InputQC_<year>);
   181→    // here we filter tanggal by prefix '<year>-'
   182→    if (filter.year) {
   183→      const yearStr = String(filter.year);
   184→      if (yearStr !== String(new Date().getFullYear())) {
   185→        where.tanggal = { ...(where.tanggal || {}), startsWith: yearStr + "-" };
   186→      }
   187→    }
   188→
   189→    const rows = await db.inputQC.findMany({
   190→      where,
   191→      orderBy: [{ tanggal: "desc" }, { inputDate: "desc" }],
   192→    });
   193→
   194→    // bidang filter needs Parameters join — apply in-memory like GAS original
   195→    let paramBidangMap: Record<string, string> = {};
   196→    if (filter.bidang) {
   197→      const params = await db.parameters.findMany({});
   198→      for (const p of params) {
   199→        paramBidangMap[p.id] = p.bidang || "Lainnya";
   200→      }
   201→    }
   202→
   203→    let result = rows.map(mapQCRow);
   204→
   205→    if (filter.bidang) {
   206→      const targetBidang = String(filter.bidang);
   207→      result = result.filter((r) => {
   208→        const b = paramBidangMap[r.paramID] || "Lainnya";
   209→        return b === targetBidang;
   210→      });
   211→    }
   212→
   213→    if (filter.namaAlat) {
   214→      const needle = String(filter.namaAlat).toLowerCase();
   215→      result = result.filter((r) => {
   216→        const hay = String(r.namaAlat || "").toLowerCase();
   217→        return hay.indexOf(needle) > -1;
   218→      });
   219→    }
   220→
   221→    return result;
   222→  } catch (e) {
   223→    return [];
   224→  }
   225→}

   227→// function getInputQCById(qcID, ownerUsername, role)
   228→// Returns single QC object or null.
   229→export async function getInputQCById(args: any[], session: SessionData | null) {
   230→  const qcID = args[0];
   231→  const ownerUsername = deriveOwner(args, session, 1);
   232→  const role = deriveRole(args, session, 2);
   233→  try {
   234→    if (!qcID) return null;
   235→    const where: any = { id: String(qcID) };
   236→    if (role !== "superadmin") where.ownerUsername = ownerUsername;
   237→    const r = await db.inputQC.findFirst({ where });
   238→    if (!r) return null;
   239→    return mapQCRow(r);
   240→  } catch (e) {
   241→    return null;
   242→  }
   243→}

   245→// function saveInputQC(payload, ownerUsername, logUser)
   246→// payload: {id?, paramID, lotID, parameter, noLot, namaAlat, tanggal,
   247→//           level1, level2, level3, catatanValidasi?}
   248→// Create (genID 'QC') or update. On edit push HistoriQC 'EDIT_QC'.
   249→export async function saveInputQC(args: any[], session: SessionData | null) {
   250→  return withLock("inputqc_save", async () => {
   251→    try {
   252→      const payload = args[0] || {};
   253→      const ownerUsername = deriveOwner(args, session, 1);
   254→      const logUser = deriveLogUser(args, session, 2);
   255→
   256→      // Resolve lot & param for autofill (mirror GAS getLotByID/getParamByID)
   257→      const [lot, param] = await Promise.all([
   258→        db.lotQC.findUnique({ where: { id: String(payload.lotID) } }),
   259→        db.parameters.findUnique({ where: { id: String(payload.paramID) } }),
   260→      ]);
   261→
   262→      const parameterStr = param ? param.parameter : String(payload.parameter || "");
   263→      const noLotStr = lot ? lot.noLot : String(payload.noLot || "");
   264→      const namaAlatStr = lot ? lot.namaAlat || null : payload.namaAlat ? String(payload.namaAlat) : null;
   265→
   266→      // tanggal: YYYY-MM-DD string
   267→      let tanggalStr: string;
   268→      if (payload.tanggal) {
   269→        const dt = parseDateStr(String(payload.tanggal));
   270→        tanggalStr = dt ? (dateToISO(dt) || new Date().toISOString().slice(0, 10)) : new Date().toISOString().slice(0, 10);
   271→      } else {
   272→        tanggalStr = new Date().toISOString().slice(0, 10);
   273→      }
   274→
   275→      const qcID = payload.id || payload.qcID || payload.qcid;
   276→
   277→      if (qcID) {
   278→        // Update path — verify ownership (case-insensitive like GAS)
   279→        const existing = await db.inputQC.findFirst({
   280→          where: {
   281→            id: String(qcID),
   282→            ownerUsername: { equals: ownerUsername },
   283→          },
   284→        });
   285→        if (!existing) {
   286→          return { ok: false, msg: "Data tidak ditemukan" };
   287→        }
   288→
   289→        // Build change-detail string (mirror GAS 'L1:old→new,L2:...,L3:...')
   290→        const changeDetail =
   291→          "L1:" + existing.level1 + "→" + payload.level1 +
   292→          ",L2:" + existing.level2 + "→" + payload.level2 +
   293→          ",L3:" + existing.level3 + "→" + payload.level3;
   294→
   295→        // Push HistoriQC row from the OLD data
   296→        await addHistoriQCInternal(existing, logUser || ownerUsername, "EDIT_QC", changeDetail);
   297→
   298→        await db.inputQC.update({
   299→          where: { id: String(qcID) },
   300→          data: {
   301→            paramID: String(payload.paramID ?? existing.paramID),
   302→            lotID: String(payload.lotID ?? existing.lotID),
   303→            parameter: parameterStr,
   304→            noLot: noLotStr,
   305→            namaAlat: namaAlatStr,
   306→            tanggal: tanggalStr,
   307→            level1: parseNumSafe(payload.level1),
   308→            level2: parseNumSafe(payload.level2),
   309→            level3: parseNumSafe(payload.level3),
   310→            catatanValidasi: payload.catatanValidasi != null ? String(payload.catatanValidasi) : existing.catatanValidasi,
   311→          },
   312→        });
   313→
   314→        await logA(ownerUsername, "EDIT_QC", String(qcID), logUser);
   315→        return { ok: true };
   316→      }
   317→
   318→      // Create path
   319→      const newID = genID("QC");
   320→      await db.inputQC.create({
   321→        data: {
   322→          id: newID,
   323→          paramID: String(payload.paramID ?? ""),
   324→          lotID: String(payload.lotID ?? ""),
   325→          parameter: parameterStr,
   326→          noLot: noLotStr,
   327→          namaAlat: namaAlatStr,
   328→          tanggal: tanggalStr,
   329→          level1: parseNumSafe(payload.level1),
   330→          level2: parseNumSafe(payload.level2),
   331→          level3: parseNumSafe(payload.level3),
   332→          inputBy: ownerUsername,
   333→          inputDate: new Date(),
   334→          validated: false,
   335→          validatedBy: null,
   336→          validatedDate: null,
   337→          catatanValidasi: payload.catatanValidasi != null ? String(payload.catatanValidasi) : null,
   338→          ownerUsername,
   339→        },
   340→      });
   341→
   342→      // GAS calls checkAndNotifyWestgard(paramID, lotID, ownerUsername, newID).
   343→      // The Westgard engine + email notifier will be ported in another chunk;
   344→      // here we just log a placeholder.
   345→      console.log(
   346→        `[inputqc] checkAndNotifyWestgard skipped for paramID=${payload.paramID} lotID=${payload.lotID} qcID=${newID}`
   347→      );
   348→
   349→      await logA(ownerUsername, "ADD_QC", newID, logUser);
   350→      return { ok: true, qcID: newID };
   351→    } catch (e: any) {
   352→      return { ok: false, msg: e.message };
   353→    }
   354→  });
   355→}

   358→// Delete QC row, push HistoriQC with actionType 'DATA DIHAPUS'.
   359→export async function deleteInputQC(args: any[], session: SessionData | null) {
   360→  return withLock("inputqc_delete", async () => {
   361→    try {
   362→      const qcID = args[0];
   363→      const ownerUsername = deriveOwner(args, session, 1);
   364→      const alasan = args[2] != null ? String(args[2]) : "";
   365→      const logUser = deriveLogUser(args, session, 3);
   366→      const by = logUser || ownerUsername;
   367→
   368→      if (!qcID) return { ok: false, msg: "Data tidak ditemukan" };
   369→
   370→      const existing = await db.inputQC.findFirst({
   371→        where: {
   372→          id: String(qcID),
   373→          ownerUsername: { equals: ownerUsername },

   375→      });
   376→      if (!existing) {
   377→        return { ok: false, msg: "Data tidak ditemukan" };

   379→
   380→      await addHistoriQCInternal(existing, by, "DATA DIHAPUS", alasan || "");

   478→export async function getQCByDateRange(args: any[], session: SessionData | null) {
   479→  try {
   480→    const paramID = args[0];

   495→    return { ok: true, data };
   496→  } catch (e: any) {
   497→    return { ok: false, msg: e.message };
   498→  }

   501→// function bulkInputQC(payload, ownerUsername, logUser)
   502→// payload: {rows: [{tanggal, level1, level2, level3}], paramID, lotID,
   503→//           parameter, noLot, namaAlat, smartPassword}
   504→// Validate smartPassword === 'didikqc'. Batch insert.
   505→// Returns {ok, count, skipped, errors}.
   506→export async function bulkInputQC(args: any[], session: SessionData | null) {
   507→  return withLock("inputqc_bulk", async () => {
   508→    try {
   509→      const payload = args[0] || {};
   510→      const ownerUsername = deriveOwner(args, session, 1);
   511→      const logUser = deriveLogUser(args, session, 2);
   512→
   513→      if (payload.smartPassword !== SMART_PWD) {
   514→        return { ok: false, msg: "Password Smart Input salah" };
   515→      }
   516→
   517→      const [lot, param] = await Promise.all([
   518→        db.lotQC.findUnique({ where: { id: String(payload.lotID) } }),
   519→        db.parameters.findUnique({ where: { id: String(payload.paramID) } }),
   520→      ]);
   521→
   522→      const parameterStr = param ? param.parameter : String(payload.parameter || "");
   523→      const noLotStr = lot ? lot.noLot : String(payload.noLot || "");
   524→      const namaAlatStr = lot ? lot.namaAlat || null : payload.namaAlat ? String(payload.namaAlat) : null;
   525→
   526→      let count = 0;
   527→      let skipped = 0;
   528→      const errors: string[] = [];
   529→      const rows: any[] = payload.rows || [];
   530→
   531→      for (const row of rows) {
   532→        if (!row.tanggal) {
   533→          errors.push("Tanggal kosong");
   534→          skipped++;
   535→          continue;
   536→        }
   537→        const dt = parseDateStr(String(row.tanggal));
   538→        if (!dt || isNaN(dt.getTime())) {
   539→          errors.push("Tanggal invalid: " + row.tanggal);
   540→          skipped++;
   541→          continue;
   542→        }
   543→        const isoDate = dateToISO(dt);
   544→        if (!isoDate) {
   545→          errors.push("Tanggal invalid: " + row.tanggal);
   546→          skipped++;
   547→          continue;
   548→        }
   549→        const l1 = parseNumSafe(row.level1);
   550→        const l2 = parseNumSafe(row.level2);
   551→        const l3 = parseNumSafe(row.level3);
   552→        if (l1 === null && l2 === null && l3 === null) {
   553→          skipped++;
   554→          continue;
   555→        }
   556→
   557→        await db.inputQC.create({
   558→          data: {
   559→            id: genID("QC"),
   560→            paramID: String(payload.paramID ?? ""),
   561→            lotID: String(payload.lotID ?? ""),
   562→            parameter: parameterStr,
   563→            noLot: noLotStr,
   564→            namaAlat: namaAlatStr,
   565→            tanggal: isoDate,
   566→            level1: l1,
   567→            level2: l2,
   568→            level3: l3,
   569→            inputBy: ownerUsername,
   570→            inputDate: new Date(),
   571→            validated: false,
   572→            validatedBy: null,
   573→            validatedDate: null,
   574→            catatanValidasi: null,
   575→            ownerUsername,
   576→          },
   577→        });
   578→        count++;
   579→      }
   580→
   581→      await logA(ownerUsername, "BULK_QC", count + " data ditambahkan", logUser);
   582→      return { ok: true, count, skipped, errors };
   583→    } catch (e: any) {
   584→      return { ok: false, msg: e.message };
   585→    }
   586→  });
   587→}

   593→// function getHistoriQC(ownerUsername, role, filter)
   594→// filter: {actionType, paramID, lotID, bidang, startDate, endDate}
   595→// Returns array of histori objects.
   596→export async function getHistoriQC(args: any[], session: SessionData | null) {
   597→  const ownerUsername = deriveOwner(args, session, 0);
   598→  const role = deriveRole(args, session, 1);
   599→  const filter = args[2] || {};
   600→  try {
   601→    const where: any = {};
   602→    if (role !== "superadmin") where.ownerUsername = ownerUsername;
   603→    if (filter.actionType) where.actionType = String(filter.actionType);
   604→    if (filter.paramID) where.paramID = String(filter.paramID);
   605→    if (filter.lotID) where.lotID = String(filter.lotID);
   606→
   607→    if (filter.startDate || filter.endDate) {
   608→      const dateFilter: any = {};
   609→      if (filter.startDate) {
   610→        const sd = parseDateStr(String(filter.startDate));
   611→        if (sd) {
   612→          const iso = dateToISO(sd);
   613→          if (iso) dateFilter.gte = iso;
   614→        }
   615→      }
   616→      if (filter.endDate) {
   617→        const ed = parseDateStr(String(filter.endDate));
   618→        if (ed) {
   619→          const iso = dateToISO(ed);
   620→          if (iso) dateFilter.lte = iso;
   621→        }
   622→      }
   623→      if (Object.keys(dateFilter).length > 0) where.tanggal = dateFilter;
   624→    }
   625→
   626→    const rows = await db.historiQC.findMany({
   627→      where,
   628→      orderBy: { deletedDate: "desc" },
   629→    });
   630→
   631→    // Build paramMap for bidang filter & parameter fallback (mirror GAS)
   632→    let paramMap: Record<string, { parameter: string; bidang: string }> = {};
   633→    if (filter.bidang) {
   634→      const params = await db.parameters.findMany({});
   635→      for (const p of params) {
   636→        paramMap[p.id] = { parameter: p.parameter, bidang: p.bidang || "Lainnya" };
   637→      }
   638→    }
   639→
   640→    let result = rows.map((r: any) => mapHistoriRow(r, paramMap[r.paramID] || null));
   641→
   642→    if (filter.bidang) {
   643→      const targetBidang = String(filter.bidang);
   644→      result = result.filter((r: any) => {
   645→        const pinfo = paramMap[r.paramID];
   646→        const b = pinfo ? pinfo.bidang : "Lainnya";
   647→        return b === targetBidang;
   648→      });
   649→    }
   650→
   651→    return result;
   652→  } catch (e) {
   653→    return [];
   654→  }
   655→}

   657→// function restoreHistoriQC(hqcID, ownerUsername)

   659→// histori row with actionType 'RESTORED'.
   660→export async function restoreHistoriQC(args: any[], session: SessionData | null) {
   661→  return withLock("inputqc_restore", async () => {
   662→    try {
   663→      const hqcID = args[0];
   664→      const ownerUsername = deriveOwner(args, session, 1);
   665→
   666→      if (!hqcID) return { ok: false, msg: "Data histori tidak ditemukan" };
   667→
   668→      // Find histori row (case-insensitive owner match — mirror GAS)
   669→      const historiRow = await db.historiQC.findFirst({
   670→        where: {
   671→          id: String(hqcID),
   672→          ownerUsername: { equals: ownerUsername },
   673→        },
   674→      });
   675→      if (!historiRow) {
   676→        return { ok: false, msg: "Data histori tidak ditemukan" };
   677→      }
   678→      if (historiRow.actionType !== "DATA DIHAPUS") {
   679→        return { ok: false, msg: "Hanya data DIHAPUS yang bisa di-restore" };
   680→      }
   681→
   682→      // Make sure the QC ID isn't already present in InputQC
   683→      if (historiRow.qcid) {
   684→        const existing = await db.inputQC.findUnique({
   685→          where: { id: String(historiRow.qcid) },
   686→        });
   687→        if (existing) {
   688→          return { ok: false, msg: "Data QC sudah ada di InputQC" };
   689→        }
   690→      }
   691→
   692→      // Recreate the InputQC row
   693→      const newQcID = historiRow.qcid || genID("QC");
   694→      await db.inputQC.create({
   695→        data: {
   696→          id: String(newQcID),
   697→          paramID: String(historiRow.paramID ?? ""),
   698→          lotID: historiRow.lotID ? String(historiRow.lotID) : "",
   699→          parameter: String(historiRow.parameter ?? ""),
   700→          noLot: historiRow.noLot ? String(historiRow.noLot) : "",
   701→          namaAlat: historiRow.namaAlat,
   702→          tanggal: historiRow.tanggal ? String(historiRow.tanggal) : new Date().toISOString().slice(0, 10),
   703→          level1: historiRow.level1,
   704→          level2: historiRow.level2,
   705→          level3: historiRow.level3,
   706→          inputBy: historiRow.inputBy || ownerUsername,
   707→          inputDate: new Date(),
   708→          validated: false,
   709→          validatedBy: null,
   710→          validatedDate: null,
   711→          catatanValidasi: null,
   712→          ownerUsername: historiRow.ownerUsername,
   713→        },
   714→      });
   715→
   716→      // Add a new histori row recording the restore
   717→      await db.historiQC.create({
   718→        data: {
   719→          id: genID("HQC"),
   720→          qcid: String(newQcID),
   721→          paramID: String(historiRow.paramID ?? ""),
   722→          lotID: historiRow.lotID,
   723→          parameter: String(historiRow.parameter ?? ""),
   724→          noLot: historiRow.noLot,
   725→          namaAlat: historiRow.namaAlat,
   726→          tanggal: historiRow.tanggal,
   727→          level1: historiRow.level1,
   728→          level2: historiRow.level2,
   729→          level3: historiRow.level3,
   730→          inputBy: historiRow.inputBy,
   731→          deletedBy: ownerUsername,
   732→          deletedDate: new Date(),
   733→          ownerUsername: historiRow.ownerUsername,
   734→          actionType: "RESTORED",
   735→          changeDetail: "Restored from HQC=" + hqcID,
   736→        },
   737→      });
   738→
   739→      await logA(ownerUsername, "RESTORE_QC", String(newQcID));
   740→      return { ok: true, msg: "Data berhasil di-restore" };

   742→      return { ok: false, msg: e.message };

   780→// function getValidasiData(ownerUsername, role, filter)
   781→// Returns QC rows enriched with z-scores per level + lot mean/SD/TEa/satuan.
   782→// z1 = (level1 - lot.meanL1) / lot.sdL1 (when lot found and SD != 0)
   783→export async function getValidasiData(args: any[], session: SessionData | null) {
   784→  const ownerUsername = deriveOwner(args, session, 0);
   785→  const role = deriveRole(args, session, 1);
   786→  const filter = args[2] || {};
   787→  try {
   788→    const qcData = await getInputQC([ownerUsername, role, filter], session);
   789→    const lots = await db.lotQC.findMany({});
   790→    const lotMap: Record<string, any> = {};
   791→    for (const l of lots) lotMap[l.id] = l;
   792→
   793→    // z-score helper (mirror GAS zS)
   794→    function zS(val: any, mean: number | null, sd: number | null): number | null {
   795→      if (val === null || val === undefined || val === "" || !mean || !sd) return null;
   796→      const v = parseNumSafe(val);
   797→      if (v === null) return null;
   798→      return (v - mean) / sd;
   799→    }
   800→
   801→    const result = qcData.map((q: any) => {
   802→      const lot = lotMap[q.lotID];
   803→      const mL1 = lot ? parseNumSafe(lot.meanL1) : null;
   804→      const sL1 = lot ? parseNumSafe(lot.sdL1) : null;
   805→      const mL2 = lot ? parseNumSafe(lot.meanL2) : null;
   806→      const sL2 = lot ? parseNumSafe(lot.sdL2) : null;
   807→      const mL3 = lot ? parseNumSafe(lot.meanL3) : null;
   808→      const sL3 = lot ? parseNumSafe(lot.sdL3) : null;
   809→      return {
   810→        qcid: q.qcid,
   811→        paramID: q.paramID,
   812→        lotID: q.lotID,
   813→        parameter: q.parameter,
   814→        noLot: q.noLot,
   815→        namaAlat: q.namaAlat,
   816→        tanggal: q.tanggal,
   817→        level1: q.level1,
   818→        level2: q.level2,
   819→        level3: q.level3,
   820→        z1: zS(q.level1, mL1, sL1),
   821→        z2: zS(q.level2, mL2, sL2),
   822→        z3: zS(q.level3, mL3, sL3),
   823→        validated: q.validated,
   824→        validatedBy: q.validatedBy,
   825→        validatedDate: q.validatedDate,
   826→        catatanValidasi: q.catatanValidasi,
   827→        lotMeanL1: mL1,
   828→        lotSDL1: sL1,
   829→        lotMeanL2: mL2,
   830→        lotSDL2: sL2,
   831→        lotMeanL3: mL3,
   832→        lotSDL3: sL3,
   833→        tea: lot ? parseNumSafe(lot.tea) : null,
   834→        satuan: lot ? lot.satuan : null,
   835→        ownerUsername: q.ownerUsername,
   836→      };
   837→    });
   838→    return result;
   839→  } catch (e) {
   840→    return [];
   841→  }
   842→}
